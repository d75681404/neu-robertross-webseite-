# VORSCHLAG – nur wirksam, wenn der Hoster Apache/.htaccess unterstützt und die Datei als ".htaccess" abgelegt wird.
# Hoster (turbopress?) ist nicht bestätigt; bitte beim Hoster klären. Nicht getestet.
<IfModule mod_headers.c>
Header always set Strict-Transport-Security "max-age=31536000"
Header always set X-Content-Type-Options "nosniff"
Header always set Referrer-Policy "strict-origin-when-cross-origin"
Header always set Permissions-Policy "camera=(), microphone=(), geolocation=()"
Header always set X-Frame-Options "SAMEORIGIN"
# CSP: Skripte/Styles nur eigen; Bilder eigen + robertross.de (noch eingebundene Altbilder); Inline-Styles/Skripte werden derzeit genutzt.
Header always set Content-Security-Policy "default-src 'self'; img-src 'self' data: https://www.robertross.de; style-src 'self' 'unsafe-inline'; script-src 'self' 'unsafe-inline'; connect-src 'self'; frame-ancestors 'self'; base-uri 'self'; form-action 'self'"
</IfModule>
